// ====== ЗВУКИ (Web Audio API) ======
let audioCtx = null;
function beep(freq, dur, type='sine') {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type; osc.frequency.value = freq;
    osc.connect(gain); gain.connect(audioCtx.destination);
    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    osc.start(); osc.stop(audioCtx.currentTime + dur);
  } catch(e){}
}
function soundCorrect() { beep(880,0.1); setTimeout(()=>beep(1320,0.15),100); }
function soundWrong() { beep(200,0.2,'square'); }
function soundWin() {
  [523,659,784,1047].forEach((f,i)=>setTimeout(()=>beep(f,0.15),i*100));
}

// ====== КОНФЕТТИ ======
function confetti() {
  const colors = ['#ff7043','#66bb6a','#42a5f5','#ffca28','#ab47bc','#ec407a'];
  for (let i = 0; i < 25; i++) {
    const c = document.createElement('div');
    c.className = 'confetti';
    c.style.left = Math.random() * 100 + '%';
    c.style.background = colors[Math.floor(Math.random()*colors.length)];
    c.style.animationDuration = (1.5 + Math.random()*1.5) + 's';
    c.style.borderRadius = Math.random() > 0.5 ? '50%' : '0';
    document.body.appendChild(c);
    setTimeout(() => c.remove(), 3000);
  }
}

  score: 0,
  streak: 0,

// ====== ОБЩИЕ ФУНКЦИИ ======
function updateHUD() {
  document.getElementById('score-display').textContent = state.score;
  document.getElementById('streak-display').textContent = state.streak;
}
function updateProgress() {
  const done = state.completed.filter(Boolean).length;
  document.getElementById('progress-fill').style.width = (done/5*100) + '%';
  const map = document.getElementById('stations-map');
  map.innerHTML = '';
  stationInfo.forEach((s, i) => {
    const d = document.createElement('div');
    d.className = 'station-dot' + (state.completed[i] ? ' done' : i === state.currentStation ? ' active' : '');
    d.textContent = state.completed[i] ? '✓' : s.icon;
    map.appendChild(d);
  });
  document.getElementById('station-title').textContent = stationInfo[state.currentStation]?.title || '';
}
function muravishkaReact(type) {
  const m = document.getElementById('muravishka');
  m.classList.remove('happy','sad');
  void m.offsetWidth;
  m.classList.add(type);
}
function rightAnswer(points = 10) {
  state.score += points;
  state.streak++;
  state.bestStreak = Math.max(state.bestStreak, state.streak);
  if (state.streak > 1) state.score += state.streak * 2; // бонус за серию
  updateHUD();
  soundCorrect();
  confetti();
  muravishkaReact('happy');
}
function wrongAnswer() {
  state.streak = 0;
  updateHUD();
  soundWrong();
  muravishkaReact('sad');
}

// ====== ИГРА ======
function startGame() {
  document.getElementById('screen-start').classList.add('hidden');
  document.getElementById('screen-game').classList.remove('hidden');
  state.currentStation = 0;
  updateHUD();
  updateProgress();
  runStation(0);
}

function nextStation() {
  document.getElementById('nextBtn').classList.add('hidden');
  state.currentStation++;
  if (state.currentStation >= 5) {
    endGame();
    return;
  }
  updateProgress();
  runStation(state.currentStation);
}

function finishStation() {
  state.completed[state.currentStation] = true;
  updateProgress();
  if (state.currentStation === 4) {
    document.getElementById('nextBtn').textContent = '🏆 Завершить';
  } else {
    document.getElementById('nextBtn').textContent = 'Дальше ➡️';
  }
  document.getElementById('nextBtn').classList.remove('hidden');
  document.getElementById('game').innerHTML += '<p style="color:#2e7d32;font-weight:bold;font-size:17px;margin-top:12px;">✅ Станция пройдена!</p>';
}

function endGame() {
  document.getElementById('screen-game').classList.add('hidden');
  document.getElementById('screen-end').classList.remove('hidden');
  document.getElementById('final-score').textContent = state.score;
  const stars = state.score >= 150 ? '⭐⭐⭐' : state.score >= 90 ? '⭐⭐' : '⭐';
  document.getElementById('final-stars').textContent = stars;
  soundWin();
  setTimeout(confetti, 300);
  setTimeout(confetti, 800);
}

function runStation(n) {
  const game = document.getElementById('game');
  game.innerHTML = '';
  if (n === 0) stationSorting();
  else if (n === 1) stationSeasons();
  else if (n === 2) stationOddOne();
  else if (n === 3) stationTrueFalse();
  else if (n === 4) stationMemory();
}

// ====== СТАНЦИЯ 1: СОРТИРОВКА ======
function stationSorting() {
  const game = document.getElementById('game');
  const items = [
    { emoji: '🌳', name: 'Берёза', alive: true },
    { emoji: '☀️', name: 'Солнце', alive: false },
    { emoji: '🐕', name: 'Собака', alive: true },
    { emoji: '🪨', name: 'Камень', alive: false },
    { emoji: '🦋', name: 'Бабочка', alive: true },
    { emoji: '🌊', name: 'Река', alive: false },
    { emoji: '🍄', name: 'Гриб', alive: true },
    { emoji: '☁️', name: 'Облако', alive: false }
  ].sort(() => Math.random() - 0.5);

  let selected = null;
  let remaining = items.length;

  game.innerHTML = `
    <h2>Распредели объекты 🧺</h2>
    <p style="font-size:14px;color:#666;">Нажми на объект, потом на корзинку</p>
    <div class="items-row" id="items-row"></div>
    <div class="baskets">
      <div class="basket green" data-alive="true">🌱<br>Живая<br>природа</div>
      <div class="basket brown" data-alive="false">🪨<br>Неживая<br>природа</div>
    </div>
  `;

  const row = document.getElementById('items-row');
  items.forEach((it, i) => {
    const el = document.createElement('div');
    el.className = 'item';
    el.textContent = it.emoji;
    el.title = it.name;
    el.dataset.id = i;
    el.onclick = () => {
      document.querySelectorAll('.item').forEach(x => x.classList.remove('selected'));
      el.classList.add('selected');
      selected = { el, item: it };
      beep(600, 0.05);
    };
    row.appendChild(el);
  });

  document.querySelectorAll('.basket').forEach(b => {
    b.onclick = () => {
      if (!selected) return;
      const isAliveBasket = b.dataset.alive === 'true';
      if (isAliveBasket === selected.item.alive) {
        rightAnswer();
        selected.el.classList.add('gone');
        setTimeout(() => selected.el.remove(), 300);
        remaining--;
        selected = null;
        if (remaining === 0) finishStation();
      } else {
        wrongAnswer();
        selected.el.classList.remove('selected');
        selected = null;
      }
    };
  });
}

// ====== СТАНЦИЯ 2: ВРЕМЕНА ГОДА ======
function stationSeasons() {
  const clues = [
    { text: '🍂 Листья желтеют и опадают', season: 'autumn' },
    { text: '❄️ На улице снег и мороз', season: 'winter' },
    { text: '🌷 Тает снег, бегут ручьи', season: 'spring' },
    { text: '☀️ Ярко светит солнце, каникулы', season: 'summer' }
  ].sort(() => Math.random() - 0.5);

  let idx = 0;
  const game = document.getElementById('game');

  function render() {
    if (idx >= clues.length) { finishStation(); return; }
    const c = clues[idx];
    game.innerHTML = `
      <h2>Времена года 🌍</h2>
      <p style="font-size:14px;color:#666;">К какому времени года относится признак?</p>
      <div class="clue-box">${c.text}</div>
      <div class="seasons-row">
        <button class="season-btn s-spring" data-s="spring"><span class="emoji">🌸</span>Весна</button>
        <button class="season-btn s-summer" data-s="summer"><span class="emoji">☀️</span>Лето</button>
        <button class="season-btn s-autumn" data-s="autumn"><span class="emoji">🍁</span>Осень</button>
        <button class="season-btn s-winter" data-s="winter"><span class="emoji">❄️</span>Зима</button>
      </div>
    `;
    game.querySelectorAll('.season-btn').forEach(b => {
      b.onclick = () => {
        game.querySelectorAll('.season-btn').forEach(x => x.disabled = true);
        if (b.dataset.s === c.season) {
          b.style.background = '#a5d6a7';
          rightAnswer();
        } else {
          b.style.background = '#ffcdd2';
          const correct = game.querySelector(`[data-s="${c.season}"]`);
          if (correct) correct.style.background = '#a5d6a7';
          wrongAnswer();
        }
        setTimeout(() => { idx++; render(); }, 900);
      };
    });
  }
  render();
}

// ====== СТАНЦИЯ 3: НАЙДИ ЛИШНЕЕ ======
function stationOddOne() {
  const rounds = [
    {
      items: ['🌳 Берёза', '🌼 Ромашка', '🌳 Дуб', '🌳 Клён'],
      correct: 1,
      explain: 'Ромашка — трава, остальные — деревья'
    },
    {
      items: ['🦊 Лиса', '🐺 Волк', '🐰 Заяц', '🐦 Воробей'],
      correct: 3,
      explain: 'Воробей — птица, остальные — звери'
    },
    {
      items: ['☀️ Солнце', '🌊 Река', '🪨 Камень', '🌳 Дерево'],
      correct: 3,
      explain: 'Дерево — живая природа, остальное — неживая'
    }
  ];
  let idx = 0;
  const game = document.getElementById('game');

  function render() {
    if (idx >= rounds.length) { finishStation(); return; }
    const r = rounds[idx];
    game.innerHTML = `<h2>Найди лишнее 🔍</h2>
      <p style="font-size:14px;color:#666;">Раунд ${idx+1} из ${rounds.length}</p>`;
    r.items.forEach((it, i) => {
      const b = document.createElement('button');
      b.className = 'option-btn';
      b.textContent = it;
      b.onclick = () => {
        game.querySelectorAll('.option-btn').forEach(x => x.disabled = true);
        if (i === r.correct) {
          b.classList.add('correct');
          rightAnswer();
        } else {
          b.classList.add('wrong');
          game.querySelectorAll('.option-btn')[r.correct].classList.add('correct');
          wrongAnswer();
        }
        const ex = document.createElement('p');
        ex.style.cssText = 'margin-top:10px;font-size:13px;color:#666;font-style:italic;';
        ex.textContent = '💡 ' + r.explain;
        game.appendChild(ex);
        setTimeout(() => { idx++; render(); }, 1600);
      };
      game.appendChild(b);
    });
  }
  render();
}

        <button class="tf-btn tf-false" data-v="false">❌ Неправда</button>
      </div>
    `;
    game.querySelectorAll('.tf-btn').forEach(b => {
      b.onclick = () => {
        game.querySelectorAll('.tf-btn').forEach(x => x.disabled = true);
        const ans = b.dataset.v === 'true';
        if (ans === f.isTrue) {
          rightAnswer();
        } else {
          wrongAnswer();
        }
        setTimeout(() => { idx++; render(); }, 900);
      };
    });
  }
  render();
}

function stationMemory() {
  const animals = ['🦊','🐻','🦉','🐸','🐝','🦋'];
  let cards = [...animals, ...animals].sort(() => Math.random() - 0.5);
  let first = null, second = null, lock = false, matched = 0;
  const game = document.getElementById('game');
  game.innerHTML = `<h2>Мемори-зоопарк 🧠</h2>
    <p style="font-size:14px;color:#666;">Найди пары животных</p>
    <div class="memory-grid" id="mem-grid"></div>
    <p style="font-size:13px;color:#666;margin-top:8px;">Пар найдено: <b id="pairs">0</b> / 6</p>`;
  const grid = document.getElementById('mem-grid');
  cards.forEach((emoji, i) => {
    const c = document.createElement('div');
    c.className = 'mem-card';
    c.textContent = '?';
    c.dataset.emoji = emoji;
    c.dataset.id = i;
    c.onclick = () => {
      if (lock || c.classList.contains('flipped') || c.classList.contains('matched')) return;
      c.classList.add('flipped');
      c.textContent = emoji;
      beep(700, 0.05);
      if (!first) { first = c; return; }
      second = c; lock = true;
      if (first.dataset.emoji === second.dataset.emoji) {
        first.classList.add('matched'); second.classList.add('matched');
        matched++;
        document.getElementById('pairs').textContent = matched;
        rightAnswer();
        first = second = null; lock = false;
        if (matched === 6) finishStation();
      } else {
        setTimeout(() => {
          first.classList.remove('flipped'); second.classList.remove('flipped');
          first.textContent = '?'; second.textContent = '?';
          first = second = null; lock = false;
          wrongAnswer();
        }, 800);
      }
    };
    grid.appendChild(c);
  });
}
